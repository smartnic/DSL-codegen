#include <immintrin.h>
#include <stdint.h>

// The widest element type is 32 bits, so a 256-bit register has 8 maximum-width lanes.
void process_packets(uint32_t *out_chain, uint32_t *out_diff, uint32_t *out_sum, uint32_t *random, uint8_t *ttl, int n) {
    {
        const int LANES = 32;
        int i = 0;
        for (; i + LANES <= n; i += LANES) {
            __m256i v = _mm256_sub_epi8(_mm256_loadu_si256((const __m256i *) (ttl + i)), _mm256_set1_epi8(1));
            _mm256_storeu_si256((__m256i *) (ttl + i), v);
        }
        for (; i < n; i++) {
            ttl[i] = (uint8_t)(ttl[i] - 1);
        }
    }
    {
        const int LANES = 32;
        int i = 0;
        for (; i + LANES <= n; i += LANES) {
            __m256i v = _mm256_sub_epi8(_mm256_loadu_si256((const __m256i *) (ttl + i)), _mm256_set1_epi8(1));
            _mm256_storeu_si256((__m256i *) (ttl + i), v);
        }
        for (; i < n; i++) {
            ttl[i] = (uint8_t)(ttl[i] - 1);
        }
    }
    {
        const int LANES = 8;
        int i = 0;
        for (; i + LANES <= n; i += LANES) {
            __m256i v = _mm256_add_epi32(_mm256_loadu_si256((const __m256i *) (random + i)), _mm256_loadu_si256((const __m256i *) (random + i)));
            _mm256_storeu_si256((__m256i *) (out_sum + i), v);
        }
        for (; i < n; i++) {
            out_sum[i] = (uint32_t)(random[i] + random[i]);
        }
    }
    {
        const int LANES = 8;
        int i = 0;
        for (; i + LANES <= n; i += LANES) {
            __m256i v = _mm256_sub_epi32(_mm256_loadu_si256((const __m256i *) (random + i)), _mm256_loadu_si256((const __m256i *) (random + i)));
            _mm256_storeu_si256((__m256i *) (out_diff + i), v);
        }
        for (; i < n; i++) {
            out_diff[i] = (uint32_t)(random[i] - random[i]);
        }
    }
    {
        const int LANES = 8;
        int i = 0;
        for (; i + LANES <= n; i += LANES) {
            __m256i v = _mm256_sub_epi32(_mm256_add_epi32(_mm256_loadu_si256((const __m256i *) (random + i)), _mm256_loadu_si256((const __m256i *) (random + i))), _mm256_loadu_si256((const __m256i *) (random + i)));
            _mm256_storeu_si256((__m256i *) (out_chain + i), v);
        }
        for (; i < n; i++) {
            out_chain[i] = (uint32_t)((uint32_t)(random[i] + random[i]) - random[i]);
        }
    }
    {
        const int LANES = 8;
        int i = 0;
        for (; i + LANES <= n; i += LANES) {
            __m256i v = _mm256_add_epi32(_mm256_loadu_si256((const __m256i *) (random + i)), _mm256_loadu_si256((const __m256i *) (random + i)));
            _mm256_storeu_si256((__m256i *) (random + i), v);
        }
        for (; i < n; i++) {
            random[i] = (uint32_t)(random[i] + random[i]);
        }
    }
}
